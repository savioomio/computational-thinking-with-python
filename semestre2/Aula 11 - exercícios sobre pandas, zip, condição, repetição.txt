import pandas as pd

df = pd.DataFrame({
    'Nomes': ['A', 'B', 'C', 'D'], 
    'Salario': [3500, 2400, 4500, 2700]})

status_salarios = []

for nome, salario in zip(df['Nomes'], df['Salario']):
  if salario > 10000:
    status_salarios.append('Alto')
  else:
    status_salarios.append('Baixo')

df['Status_Salario'] = status_salarios
